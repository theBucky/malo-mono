// 统计每个字符出现的次数
type Counts = Map<string, number>;

export function countChars(text: string): Counts {
  const counts: Counts = new Map();
  for (const char of text) {
    counts.set(char, (counts.get(char) ?? 0) + 1);
  }
  return counts;
}
